8. 실제 브랜드 19개 등록하기

from django.db import transaction
from brands.models import Brand

with transaction.atomic():
    for brand in brands:
        Brand.objects.update_or_create(
            slug=brand['id'],
            defaults={
                'name': brand['name'],
                'category': brand['category'],
                'tagline': brand.get('tagline') or '',
                'english_name': brand.get('english_name') or '',
                'image': brand.get('image') or 'coffee.jpg',
                'is_demo': False,
            }
        )

코드 설명
- transaction.atomic() : 등록 도중 오류가 발생하면 이번 작업의 변경 사항을 되돌려 줘.
- for brand in brands : JSON에 있는 19개 브랜드를 하나씩 가져와.
- update_or_create() : 같은 slug가 있으면 수정하고, 없으면 새로 등록해.
- brand.get('tagline') or '' : 설명이 없으면 빈 문자열로 저장해.
- is_demo: False : 실제 브랜드 데이터임을 표시해.

10.실제 연도별 통계 48개 등록하기

AnnualStat 모델에 데이터를 저장

from django.db import transaction
from brands.models import Brand, AnnualStat

with transaction.atomic():
    for brand_data in brands:
        brand = Brand.objects.get(slug=brand_data['id'])

        for history in brand_data['history']:
            AnnualStat.objects.update_or_create(
                brand=brand,
                year=history['year'],
                defaults={
                    'stores': history.get('stores'),
                    'opened': history.get('opened'),
                    'closed': history.get('closed'),
                    'annual_sales_krw': history.get('annual_sales_krw'),
                    'source_label': '브랜드별 수집 데이터',
                    'source_url': '',
                }
            )

코드 설명
- Brand.objects.get() : 앞에서 등록한 실제 브랜드를 찾는 코드야.
- for history in brand_data['history'] : 해당 브랜드의 2024~2026년 데이터를 하나씩 가져와.
- AnnualStat.objects.update_or_create() : 브랜드와 연도가 같으면 수정하고, 없으면 새로 등록해.
- history.get() : JSON 값이 null이면 Python의 None으로 저장해.
- transaction.atomic() : 작업 중 오류가 발생하면 이번 저장 작업 전체를 되돌려 줘.
source_label은 우선 임시로 표시하는 값이야. 나중에 실제 출처 정보를 연결할 거야.

주의: JSON의 year와 source_accounting_year가 다를 수 있으므로, 현재는 JSON에 표시된 year를 기준으로 저장해. 원자료 기준 연도는 추후 출처 정리 시 별도로 확인

12. 실제 지역별 통계 DB 등록

from django.db import transaction
from brands.models import Brand, AnnualStat, RegionalStat

with transaction.atomic():
    for brand_data in brands:
        brand = Brand.objects.get(slug=brand_data['id'])

        for history in brand_data['history']:
            annual_stat = AnnualStat.objects.get(
                brand=brand,
                year=history['year']
            )

            regions = history.get('regions') or {}

            for region_name, region_data in regions.items():
                RegionalStat.objects.update_or_create(
                    annual_stat=annual_stat,
                    region=region_name,
                    defaults={
                        'stores': region_data.get('stores')
                    }
                )


코드			역할
transaction.atomic()	오류 발생 시 이번 저장 작업을 되돌림
Brand.objects.get()		실제 브랜드 조회
AnnualStat.objects.get()	해당 브랜드의 연도별 통계 조회
history.get('regions') or {}	지역 데이터가 없으면 빈 딕셔너리 사용
regions.items()		지역명과 지역 데이터를 하나씩 가져옴
update_or_create()		기존 데이터가 있으면 수정, 없으면 생성
region_data.get('stores')	해당 지역의 가맹점 수 저장

16. 실제 창업비용 DB 등록하기

from django.db import transaction
from brands.models import Brand

with transaction.atomic():
    for brand_data in brands:

        # 창업비용이 있는 연도별 자료만 가져오기
        valid_history = [
            h for h in brand_data['history']
            if h.get('startup_cost_krw') is not None
        ]

        # 가장 최근 연도 선택
        latest = max(
            valid_history,
            key=lambda h: h['year'],
            default=None
        )

        # DB에서 실제 브랜드 조회
        brand = Brand.objects.get(
            slug=brand_data['id'],
            is_demo=False
        )

        if latest is not None:
            brand.initial_cost_krw = latest['startup_cost_krw']

            brand.costs = {
                'year': latest['year'],
                'startup_cost_breakdown': (
                    latest.get('startup_cost_breakdown') or {}
                ),
                'history': [
                    {
                        'year': h['year'],
                        'startup_cost_krw': h.get('startup_cost_krw'),
                        'startup_cost_breakdown': (
                            h.get('startup_cost_breakdown') or {}
                        )
                    }
                    for h in brand_data['history']
                ],
                'startup_cost_offers': (
                    brand_data.get('startup_cost_offers') or []
                )
            }

            brand.save(
                update_fields=['initial_cost_krw', 'costs']
            )


코드 설명
- valid_history : 창업비용이 None이 아닌 연도별 자료만 가져와.
- max(..., key=lambda h: h['year']) : 가장 최근 연도의 비용을 선택해.
- initial_cost_krw : 사이트에서 대표적으로 표시할 창업비용을 저장해.
- costs : 연도별 비용과 세부 항목, 별도 수집 자료를 JSON 형태로 보존해.
- transaction.atomic() : 저장 중 오류가 발생하면 이번 작업 전체를 취소해.
주의: 별도 수집 자료인 startup_cost_offers는 보관만 하고, 대표 창업비용을 덮어쓰지 않아.

22. SQLite DB 실제 백업하기

import sqlite3

# 원본 DB 연결
source = sqlite3.connect(str(db_path))

# 백업 DB 연결
destination = sqlite3.connect(str(backup_path))

try:
    # SQLite 백업 기능 실행
    source.backup(destination)
finally:
    destination.close()
    source.close()

print('백업 완료!')
print('백업 파일:', backup_path)
print('백업 파일 존재:', backup_path.exists())
print('백업 파일 크기:', backup_path.stat().st_size, 'bytes')

코드 설명
- sqlite3.connect() : 원본 DB와 백업 DB를 연결해.
- source.backup(destination) : 원본 SQLite 데이터를 백업 파일로 복사해.
- finally : 백업 성공 여부와 관계없이 연결을 종료해.
- backup_path.exists() : 백업 파일이 실제로 생성됐는지 확인해.
- backup_path.stat().st_size : 백업 파일 크기를 확인해.

def visible_brands():
    # 실제 브랜드가 들어오면 예시 브랜드를 목록에서 제외합니다.
    qs = Brand.objects.all()
    if qs.filter(is_demo=False).exists():
        qs = qs.filter(is_demo=False)
    return qs.prefetch_related('annual_stats__regional_stats').order_by('id')

35.money() 함수 수정
이전것 주석처리

@register.filter
def money(value):
    # 값이 없으면 미제공
    if value is None or value == '':
        return '미제공'

    # 딕셔너리는 숫자로 계산할 수 없으므로 별도 처리
    if isinstance(value, dict):
        return '미제공'

    # 숫자가 아닌 값도 안전하게 처리
    try:
        return f'{round(float(value) / 10000):,}만원'
    except (TypeError, ValueError):
        return '미제공'

전달된 값			출력 결과
130491000		13,049만원
None			미제공
''			미제공
{'가맹비': 10000000}	미제공

주의: 이 수정은 오류를 방지하는 임시 조치야. 딕셔너리 안에 실제 창업비용이 있더라도 미제공으로 표시되므로, 데이터가 정확하게 출력되는 최종 해결 방법은 아니야.

def brand_row(brand, year, region='전국'):
    # 선택 연도에 해당하는 통계를 찾습니다. 없으면 None입니다.
    stat = None
    for annual in brand.annual_stats.all():
        if annual.year == year:
            stat = annual
            break
    region_counts = {}
    monthly = None
    if stat:
        for regional in stat.regional_stats.all():
            region_counts[regional.region] = regional.stores
        if stat.annual_sales_krw is not None:
            monthly = round(stat.annual_sales_krw / 12 / 10000)
    # 전국 매장 수 계산 : 디장고, 파이썬에선 None, json에서는 null
    if stat and region == '전국':
        total_stores = stat.total_stores       # brands2.json의 total_stores (전체 매장 수)
        franchise_stores = stat.stores         # brands2.json의 stores (가맹점 수)
        direct_stores = stat.direct_stores     # brands2.json의 direct_stores (직영점 수)

        # 전체 매장 수가 None이 아니면 해당 값을 우선 사용
        if total_stores is not None:
            stores = total_stores

        # 전체 매장 수가 없고, 가맹점 또는 직영점 수 중 하나라도 값이 있으면 합산
        # 계산할 때 None은 0으로 취급하지만, DB에 저장된 None은 변경하지 않음
        elif franchise_stores is not None or direct_stores is not None:
            stores = (franchise_stores or 0) + (direct_stores or 0)

        # 전체 매장 수, 가맹점 수, 직영점 수가 모두 없으면 None
        else:
            stores = None

    # 전국이 아닌 경우 기존 지역별 가맹점 수 사용
    else:
        stores = region_counts.get(region)
    try:
        area_record = stat.area_sales if stat else None
    except ObjectDoesNotExist:
        area_record = None
    per_pyeong = area_record.annual_sales_per_pyeong_krw if area_record else None
    return {'brand': brand, 'stat': stat, 'monthly': monthly, 'stores': stores, 'region_counts': region_counts,
            'area_pyeong': round(brand.area_sqm / 3.305785, 1),

            'per_pyeong_annual': round(per_pyeong / 10000, 2) if per_pyeong is not None else None,

            'area_source': area_record.source_label if area_record else ''}
